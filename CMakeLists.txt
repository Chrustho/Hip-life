cmake_minimum_required(VERSION 3.21)
project(hip-life LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

option(HIPLIFE_ENABLE_HIP "Build the HIP GPU backend" ON)
option(HIPLIFE_CPU_REFERENCE "Build CPU reference and GPU-vs-CPU comparison tests" ON)
set(HIPLIFE_GFX_ARCH "gfx1152" CACHE STRING "GPU offload target for hipcc (--offload-arch)")

set(HL_HAVE_HIP OFF)
if(HIPLIFE_ENABLE_HIP)
    include(CheckLanguage)
    check_language(HIP)
    if(CMAKE_HIP_COMPILER)
        enable_language(HIP)
        set(HL_HAVE_HIP ON)
    else()
        message(STATUS "hip-life: no HIP compiler found, building CPU-only")
    endif()
endif()

set(HL_CORE_SOURCES src/sim/cpu_simulation.cpp)
if(HL_HAVE_HIP)
    list(APPEND HL_CORE_SOURCES src/sim/life.hip)
else()
    list(APPEND HL_CORE_SOURCES src/sim/hip_unavailable.cpp)
endif()

add_library(hiplife_core STATIC ${HL_CORE_SOURCES})
target_include_directories(hiplife_core PUBLIC src)
if(HL_HAVE_HIP)
    set_source_files_properties(src/sim/life.hip PROPERTIES LANGUAGE HIP)
    set_target_properties(hiplife_core PROPERTIES HIP_ARCHITECTURES "${HIPLIFE_GFX_ARCH}")
    target_compile_definitions(hiplife_core PUBLIC HIPLIFE_HIP)
endif()

find_package(raylib QUIET)
if(NOT raylib_FOUND)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(RAYLIB IMPORTED_TARGET raylib)
    endif()
endif()

add_executable(hip-life src/main.cpp)
target_include_directories(hip-life PRIVATE src)
if(raylib_FOUND)
    target_link_libraries(hip-life PRIVATE raylib)
else()
    target_link_libraries(hip-life PRIVATE PkgConfig::RAYLIB)
endif()

enable_testing()
add_executable(hiplife_invariants tests/invariants.cpp)
target_link_libraries(hiplife_invariants PRIVATE hiplife_core)
add_test(NAME invariants COMMAND hiplife_invariants)

if(HL_HAVE_HIP AND HIPLIFE_CPU_REFERENCE)
    add_executable(hiplife_gpu_vs_cpu tests/gpu_vs_cpu.cpp)
    target_link_libraries(hiplife_gpu_vs_cpu PRIVATE hiplife_core)
    add_test(NAME gpu_vs_cpu COMMAND hiplife_gpu_vs_cpu)
endif()
