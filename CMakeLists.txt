cmake_minimum_required(VERSION 3.21)
project(hip-life LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_library(hiplife_core STATIC src/sim/cpu_simulation.cpp src/sim/hip_unavailable.cpp)
target_include_directories(hiplife_core PUBLIC src)

find_package(raylib QUIET)
if(NOT raylib_FOUND)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(RAYLIB IMPORTED_TARGET raylib)
    endif()
endif()

add_executable(hip-life src/main.cpp)
target_include_directories(hip-life PRIVATE src)
if(raylib_FOUND)
    target_link_libraries(hip-life PRIVATE raylib)
else()
    target_link_libraries(hip-life PRIVATE PkgConfig::RAYLIB)
endif()

enable_testing()
add_executable(hiplife_invariants tests/invariants.cpp)
target_link_libraries(hiplife_invariants PRIVATE hiplife_core)
add_test(NAME invariants COMMAND hiplife_invariants)
